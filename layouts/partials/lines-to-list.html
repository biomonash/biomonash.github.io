{{- /* Turns a Notion value into a list for {{ range }}. Accepts: - Multi-select (already a list) - Text written as a list: ["first item", "second item"] - Text with one item per line Blank items and leading "•" / "-" bullets are removed. Usage: {{ $items := partial "lines-to-list.html" .Params.something }} */ -}} {{- $out := slice -}} {{- $items := slice -}} {{- with . -}} {{- if reflect.IsSlice . -}} {{- $items = . -}} {{- else -}} {{- $text := strings.TrimSpace (replace (string .) "\r" "") -}} {{- $parsed := false -}} {{- if and (hasPrefix $text "[") (hasSuffix $text "]") -}} {{- with try (transform.Unmarshal (dict "format" "json") $text) -}} {{- if not .Err -}} {{- $items = .Value -}} {{- $parsed = true -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $parsed -}} {{- $items = split $text "\n" -}} {{- end -}} {{- end -}} {{- end -}} {{- range $items -}} {{- $line := strings.TrimLeft "•-* " (strings.TrimSpace (string .)) -}} {{- with $line -}} {{- $out = $out | append . -}} {{- end -}} {{- end -}} {{- return $out -}}
